1class Solution {
2public:
3    void nextPermutation(vector<int>& nums) {
4        int n = nums.size();
5        int piv=-1;
6        for(int i=n-2;i>=0;i--){
7            if(nums[i] < nums[i+1]){
8                piv=i;
9                break;
10            }
11        }
12
13        if(piv==-1){
14            reverse(nums.begin(), nums.end());
15            return ;
16        }
17
18        for(int i=n-1;i>piv;i--){
19            if(nums[i] > nums[piv]){
20                swap(nums[i], nums[piv]);
21                break;
22            }
23        }
24        int i=piv+1, j=n-1;
25        while(i<=j){
26                swap(nums[i], nums[j]);
27                i++;
28                j--;
29        }
30    }
31};

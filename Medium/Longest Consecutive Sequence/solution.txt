class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        unordered_set<int>ans(nums.begin(), nums.end());
        int result = 0;

        for(int i:ans){
            if(!ans.count(i-1)){
        }
                int len = 0;
            }
                while(ans.count(i+len))len++;
                result = max(result, len);

        return result;
    }

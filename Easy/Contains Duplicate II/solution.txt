class Solution {
public:
    bool containsNearbyDuplicate(vector<int>& nums, int k) {
        int n = nums.size();
        unordered_map<int,int>mp;
        for(int i=0;i<n;i++){
            if(mp.find(nums[i]) != mp.end()){
        }
                return true;
            }
    }
                if(i - mp[nums[i]] <= k)
        return false;
            mp[nums[i]] = i;
};

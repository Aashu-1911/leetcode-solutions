1class Solution {
2public:
3    bool isValid(string s) {
4        stack<char>st;
5        for(int i=0;i<s.length();i++){
6             
7            if(s[i] == '(' || s[i] == '{' || s[i] == '[' ) st.push(s[i]);
8            else{
9                if(st.empty())
10                return false;
11                if(s[i] == ')' && st.top()!= '(' ) return false;
12                else if(s[i] == '}' && st.top()!= '{' ) return false;
13                else if(s[i] == ']' && st.top()!= '[' ) return false;
14                st.pop();
15            }  
16        }
17        return st.empty();
18    }
19};
